--[[
================================================================
  QUY AUTO  (Auto Lock-On + Auto Defense - LocalScript duy nhat)
----------------------------------------------------------------
  1) AUTO LOCK: danh thuong / ky nang -> ghim quai gan nhat trong tam,
     nhan vat + camera quay ve muc tieu, het ky nang camera tra ve.
  2) AUTO DEFENSE: don danh / ky nang sap cham nguoi choi ->
       - don DO DUOC  -> tu dong do (giu do), het don thi nha do
       - don KHONG DO DUOC -> tu dong ne (sidestep)
       - nhieu don cung luc -> phong thu don co SAT THUONG CAO NHAT
  3) Nut tron nho: bam de hien / an menu (keo di duoc)

  DAT: StarterPlayer > StarterPlayerScripts (LocalScript)

  DE AUTO DEFENSE HOAT DONG, game cua ban can:
   - Gan tag "EnemyAttack" (CollectionService) cho hitbox / projectile
     cua don danh quai. (co the doi ten trong CONFIG.Defense.AttackTag)
   - Tren hitbox (hoac model quai) dat Attribute:
       Damage    (number)  sat thuong cua don
       Blockable (bool)    true = do duoc, false = phai ne
       Active    (bool)    tuy chon, false = hitbox chua bat
   - Hanh dong do: RemoteEvent ten "BlockRemote" trong ReplicatedStorage
     (FireServer(true/false)) hoac dien vao CONFIG.Defense.OnBlock
   - Hanh dong ne: mac dinh buoc ngang ne (local). Muon dung ky nang ne
     cua game thi dat OnDodge / DodgeRemoteName va LocalDodge = false
================================================================
--]]

local NotifyContainer
local function notify(text, isError)
    if not NotifyContainer or not NotifyContainer.Parent then return end
    local frame = Instance.new("Frame")
    frame.BackgroundColor3 = isError and Color3.fromRGB(180,60,60) or Color3.fromRGB(40,45,55)
    frame.BackgroundTransparency = 0.1
    frame.Size = UDim2.new(0, 200, 0, 26)
    frame.AutomaticSize = Enum.AutomaticSize.X
    frame.Position = UDim2.new(0.5, 0, 0, 0)
    frame.AnchorPoint = Vector2.new(0.5, 0)
    frame.BorderSizePixel = 0
    Instance.new("UICorner", frame).CornerRadius = UDim.new(0, 6)
    local l = Instance.new("TextLabel")
    l.BackgroundTransparency = 1
    l.Size = UDim2.new(1,-16,1,0); l.Position = UDim2.new(0,8,0,0)
    l.Font = Enum.Font.GothamMedium; l.TextSize = 11
    l.TextColor3 = Color3.new(1,1,1); l.Text = text; l.Parent = frame
    frame.Parent = NotifyContainer
    task.spawn(function()
        task.wait(CONFIG.NotifyDuration)
        if frame.Parent then
            TweenService:Create(frame, TweenInfo.new(0.3), {BackgroundTransparency=1}):Play()
            TweenService:Create(l, TweenInfo.new(0.3), {TextTransparency=1}):Play()
            task.wait(0.35); frame:Destroy()
        end
    end)
    for _, o in ipairs(NotifyContainer:GetChildren()) do
        if o:IsA("Frame") and o ~= frame then
            local y = o.Position.Y.Offset
            TweenService:Create(o, TweenInfo.new(0.2),
                {Position = UDim2.new(0.5,0,0,y-32)}):Play()
        end
    end
end

local function getEnemyRoot(model)
    if not model or not model:IsA("Model") then return nil end
    return model:FindFirstChild("HumanoidRootPart")
        or model.PrimaryPart
        or model:FindFirstChildWhichIsA("BasePart", true)
end

local function isValidEnemy(model)
    if not model or not model.Parent then return false end
    if not model:IsA("Model") then return false end
    if model == getCharacter() then return false end
    if Players:GetPlayerFromCharacter(model) then return false end
    local hum = model:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 then return false end
    return getEnemyRoot(model) ~= nil
end

local function collectEnemies()
    local list, seen = {}, {}
    local function add(m)
        if not seen[m] and isValidEnemy(m) then
            seen[m] = true
            table.insert(list, m)
        end
    end
    for _, m in ipairs(CollectionService:GetTagged(CONFIG.EnemyTag)) do add(m) end
    local folder = workspace:FindFirstChild(CONFIG.FallbackEnemyFolder)
    if folder then
        for _, m in ipairs(folder:GetChildren()) do add(m) end
    end
    return list
end

local function findNearestEnemy(range)
    local hrp = getHRP()
    if not hrp then return nil end
    local best, bestDist = nil, range
    for _, m in ipairs(collectEnemies()) do
        local root = getEnemyRoot(m)
        local d = (root.Position - hrp.Position).Magnitude
        if d <= bestDist then
            best, bestDist = m, d
        end
    end
    return best
end

local function refreshStatus()

local function setStatus(text)
    if Defense.status ~= text then
        Defense.status = text
        refreshStatus()
    end

local function attrOf(inst, name)
    local v = inst:GetAttribute(name)
    if v ~= nil then return v end
    local m = inst:FindFirstAncestorOfClass("Model")
    if m then return m:GetAttribute(name) end
    return nil
end

local function fireRemote(name, ...)
    if not name then return end
    local r = ReplicatedStorage:FindFirstChild(name)
    if r and r:IsA("RemoteEvent") then r:FireServer(...) end
end

local blockFx, warnedBlock = nil, false
local function setBlock(on)
    if Defense.blocking == on then return end
    Defense.blocking = on
    local D = CONFIG.Defense
    fireRemote(D.BlockRemoteName, on)
    if D.OnBlock then pcall(D.OnBlock, on) end

    -- hieu ung xanh tren nhan vat = bang chung trang thai DO dang hoat dong
    if blockFx then blockFx:Destroy(); blockFx = nil end
    if on then
        local char = getCharacter()
        if char then
            blockFx = Instance.new("Highlight")
            blockFx.FillColor = Color3.fromRGB(80, 160, 255)
            blockFx.FillTransparency = 0.6
            blockFx.OutlineColor = Color3.fromRGB(180, 220, 255)
            blockFx.Parent = char
        end
        local hasRemote = D.BlockRemoteName and ReplicatedStorage:FindFirstChild(D.BlockRemoteName)
        if not hasRemote and not D.OnBlock and not warnedBlock then
            warnedBlock = true
            notify("Chua noi hanh dong DO cua game (BlockRemote/OnBlock): chi co hieu ung xanh", true)
        end
    end
end

local function doDodge(th)
    local D = CONFIG.Defense
    local now = os.clock()
    if now < Defense.nextDodge then return end
    local hrp = getHRP()
    if not hrp then return end
    local hum = getHumanoid()
    local mv = hum and hum.MoveDirection or Vector3.zero

    local relFlat = Vector3.new(hrp.Position.X - th.pos.X, 0, hrp.Position.Z - th.pos.Z)
    local travel  = Vector3.new(th.vel.X, 0, th.vel.Z)
    local vec
    if travel.Magnitude > 2 then
        -- don dang bay: ne vuong goc voi duong bay, ra phia nguoi choi dang dung
        travel = travel.Unit
        local side = Vector3.new(-travel.Z, 0, travel.X)
        local s = relFlat:Dot(side)
        if math.abs(s) < 0.5 then s = mv:Dot(side) end
        vec = side * (s >= 0 and 1 or -1)
    else
        -- vung sat thuong dung yen: chay thang ra xa tam
        vec = relFlat.Magnitude > 0.05 and relFlat.Unit or -hrp.CFrame.LookVector
    end

    Defense.nextDodge = now + D.DodgeCooldown
    Defense.dodgeUntil = now + D.DodgeTime
    Defense.dodgeVec = vec
    if D.LocalDodge then
        local ok = pcall(function()
            local att = Instance.new("Attachment")
            att.Parent = hrp
            local lv = Instance.new("LinearVelocity")
            lv.Attachment0 = att
            lv.RelativeTo = Enum.ActuatorRelativeTo.World
            lv.VelocityConstraintMode = Enum.VelocityConstraintMode.Vector
            lv.ForceLimitMode = Enum.ForceLimitMode.PerAxis
            lv.MaxAxesForce = Vector3.new(1e6, 0, 1e6)   -- truc Y de trong: van roi tu do
            lv.VectorVelocity = vec * D.DodgeSpeed
            lv.Parent = hrp
            task.delay(D.DodgeTime, function() lv:Destroy(); att:Destroy() end)
        end)
        if not ok then
            hrp.AssemblyLinearVelocity = Vector3.new(vec.X * D.DodgeSpeed,
                hrp.AssemblyLinearVelocity.Y, vec.Z * D.DodgeSpeed)
        end
    end
    fireRemote(D.DodgeRemoteName, vec)
    if D.OnDodge then pcall(D.OnDodge, vec) end
end

-- Van toc hitbox: neu game di chuyen hitbox bang CFrame/Tween (van toc = 0) thi tu uoc luong
local lastSeen = setmetatable({}, { __mode = "k" })
local function getPartVelocity(part, now)
    local vel = part.AssemblyLinearVelocity
    local pos = part.Position
    local prev = lastSeen[part]
    lastSeen[part] = { pos = pos, t = now }
    if vel.Magnitude < 1 and prev then
        local dt = now - prev.t
        if dt >= 0.015 and dt <= 0.3 then
            local est = (pos - prev.pos) / dt
            if est.Magnitude < 500 then return est end
        end
    end
    return vel
end

-- Mo phong duong di cua hitbox va nguoi choi trong ReactTime giay toi.
-- Tra ve thoi diem cham dau tien, hoac nil neu KHONG TRUNG.
local function timeToHit(part, vel, hrpPos, hrpVel, D)
    local cf = part.CFrame
    local isBall = part:IsA("Part") and part.Shape == Enum.PartType.Ball
    local pad = Vector3.new(D.HitPadding, D.HitPadding, D.HitPadding)
    local half = part.Size / 2 + D.PlayerHalfSize + pad
    local radius = part.Size.X / 2 + math.max(D.PlayerHalfSize.X, D.PlayerHalfSize.Z) + D.HitPadding
    local minHalf = math.max(0.5, math.min(half.X, half.Y, half.Z))
    local relSpeed = (vel - hrpVel).Magnitude
    -- buoc thoi gian nho khi don nhanh de khong "xuyen qua" hitbox
    local step = math.clamp(minHalf * 0.8 / math.max(relSpeed, 1), 0.005, 0.05)

    local t = 0
    while t <= D.ReactTime do
        local rel = (hrpPos + hrpVel * t) - (cf.Position + vel * t)
        if isBall then
            if rel.Magnitude <= radius then return t end
        else
            local l = cf:VectorToObjectSpace(rel)
            if math.abs(l.X) <= half.X and math.abs(l.Y) <= half.Y and math.abs(l.Z) <= half.Z then
                return t
            end
        end
        t += step
    end
    return nil
end

-- Chi tinh cac don SE TRUNG nguoi choi; chon don SAT THUONG CAO NHAT
local function scanThreats(now)
    local hrp = getHRP()
    local char = getCharacter()
    if not hrp or not char then return nil end
    local D = CONFIG.Defense
    local hrpPos, hrpVel = hrp.Position, hrp.AssemblyLinearVelocity
    local best

    for _, part in ipairs(CollectionService:GetTagged(D.AttackTag)) do
        if part:IsA("BasePart") and part:IsDescendantOf(workspace)
            and not part:IsDescendantOf(char) and attrOf(part, "Active") ~= false then

            local vel = getPartVelocity(part, now)
            local t = timeToHit(part, vel, hrpPos, hrpVel, D)
            if t then
                local dmg = tonumber(attrOf(part, "Damage")) or D.DefaultDamage
                if not best or dmg > best.damage or (dmg == best.damage and t < best.t) then
                    best = {
                        part = part,
                        pos = part.Position,
                        vel = vel,
                        t = t,
                        damage = dmg,
                        blockable = attrOf(part, "Blockable") ~= false,
                    }
                end
            end
        end
    end
    return best
end

local scanAcc, diagAcc = 0, 0
track(RunService.Heartbeat:Connect(function(dt)
    local D = CONFIG.Defense
    local now = os.clock()
    local hrp = getHRP()

    -- chan doan: dem so quai va so hitbox script dang thay (khi menu mo)
    diagAcc += dt
    if diagAcc >= 0.5 then
        diagAcc = 0
        if menuFrame and menuFrame.Visible then
            diagText = (" | Quai %d | Don %d"):format(#collectEnemies(),
                #CollectionService:GetTagged(D.AttackTag))
            refreshStatus()
        end
    end

    if not Defense.enabled or not isAlive() then
        Defense.reacting = false
        if Defense.blocking then setBlock(false) end
        setStatus(Defense.enabled and "Ranh" or "Tat")
        return
    end

    scanAcc += dt
    if scanAcc < D.ScanInterval then return end
    scanAcc = 0

    local th = scanThreats(now)
    if th then
        Defense.lastThreat = now
        Defense.reacting = true
        Defense.threatPart = th.part
        Defense.threatPos = th.pos
        Defense.camAim = th.pos
        AutoLock:interrupt()                     -- bo ghim quai NGAY LAP TUC
        if th.blockable then
            setBlock(true)                       -- don do duoc -> do
            setStatus("Do")
        else
            if Defense.blocking then setBlock(false) end
            doDodge(th)                          -- don khong do duoc -> ne
            setStatus("Ne")
        end
    elseif now - Defense.lastThreat >= D.ReleaseDelay then
        -- khong con don nao se trung -> nha do, tra camera
        Defense.reacting = false
        Defense.threatPart = nil
        if Defense.blocking then setBlock(false) end
        setStatus("Ranh")
    end
end))

AutoLock = {
    enabled = true,
    target = nil,
    expire = 0,
    weight = 0,
    lastAimPos = nil,
    savedAutoRotate = nil,
    highlight = nil,
    lockRange = 40,
    startTime = 0,
}

local function setHighlight(model)
    if AutoLock.highlight then
        AutoLock.highlight:Destroy()
        AutoLock.highlight = nil
    end
    if model then
        local h = Instance.new("Highlight")
        h.FillTransparency = 1
        h.OutlineColor = Color3.fromRGB(255, 60, 60)
        h.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
        h.Parent = model
        AutoLock.highlight = h
    end
end

local function restoreAutoRotate()
    if AutoLock.savedAutoRotate ~= nil then
        local hum = getHumanoid()
        if hum then hum.AutoRotate = AutoLock.savedAutoRotate end
        AutoLock.savedAutoRotate = nil
    end
end

local function releaseTarget()
    AutoLock.target = nil
    AutoLock.expire = 0
    setHighlight(nil)
    restoreAutoRotate()
end

-- Bi tan cong: bo ghim tuc thi, camera do Defense dieu khien
function AutoLock:interrupt()
    if self.target then releaseTarget() end
    self.weight = 0
    self.lastAimPos = nil
end

function AutoLock:trigger(duration, range)
    if not self.enabled or not isAlive() or Defense.reacting then return end
    range = range or CONFIG.LockRange
    local target = findNearestEnemy(range)
    if not target then return end
    if target ~= self.target then
        self.target = target
        setHighlight(target)
    end
    local now = os.clock()
    self.startTime = now
    self.expire = now + duration
    self.lockRange = range
end

RunService:BindToRenderStep("AutoLockCamera", Enum.RenderPriority.Camera.Value + 1, function(dt)
    local cam = workspace.CurrentCamera
    local hrp = getHRP()
    local now = os.clock()
    local active = false

    local target = AutoLock.target
    if target then
        local root = isValidEnemy(target) and getEnemyRoot(target) or nil
        if root and hrp and isAlive() and now < AutoLock.expire
            and (root.Position - hrp.Position).Magnitude <= AutoLock.lockRange * CONFIG.LoseRangeMul then
            active = true
            AutoLock.lastAimPos = root.Position

            if hrp.AssemblyLinearVelocity.Magnitude >= CONFIG.DashSpeed then
                local cap = AutoLock.startTime + CONFIG.MaxLockTime
                AutoLock.expire = math.max(AutoLock.expire, math.min(now + CONFIG.DashGrace, cap))
            end

            -- dang do thi uu tien quay ve phia don, khong quay ve quai
            if CONFIG.TurnCharacter and not Defense.blocking then
                local hum = getHumanoid()
                if hum then
                    if AutoLock.savedAutoRotate == nil then
                        AutoLock.savedAutoRotate = hum.AutoRotate
                    end
                    hum.AutoRotate = false
                end
                local flat = Vector3.new(root.Position.X, hrp.Position.Y, root.Position.Z)
                if (flat - hrp.Position).Magnitude > 0.05 then
                    hrp.CFrame = CFrame.lookAt(hrp.Position, flat)
                end
            end
        else
            releaseTarget()  -- het ky nang / quai chet / ra khoi tam
        end
    end

    if active then
        AutoLock.weight = math.min(1, AutoLock.weight + dt / CONFIG.CamBlendIn)
    else
        AutoLock.weight = math.max(0, AutoLock.weight - dt / CONFIG.CamBlendOut)
    end

    if AutoLock.weight > 0 and AutoLock.lastAimPos and cam then
        local cf = cam.CFrame
        if (AutoLock.lastAimPos - cf.Position).Magnitude > 0.1 then
            local look = CFrame.lookAt(cf.Position, AutoLock.lastAimPos)
            cam.CFrame = cf:Lerp(look, AutoLock.weight)
        end
    end

    -- ===== AUTO DEFENSE: uu tien cao nhat, de len ghim =====
    local D = CONFIG.Defense
    local part = Defense.threatPart
    if Defense.reacting and part and part.Parent then
        Defense.camAim = part.Position          -- bam theo don dang bay toi
    end
    if Defense.reacting and Defense.camAim then
        Defense.camWeight = math.min(1, Defense.camWeight + dt / D.CamSnapIn)
    else
        Defense.camWeight = math.max(0, Defense.camWeight - dt / CONFIG.CamBlendOut)
    end
    if Defense.camWeight > 0 and Defense.camAim and cam then
        local cf = cam.CFrame
        if (Defense.camAim - cf.Position).Magnitude > 0.1 then
            cam.CFrame = cf:Lerp(CFrame.lookAt(cf.Position, Defense.camAim), Defense.camWeight)
        end
    end

    -- dang do: nhan vat quay mat ve phia don
    if Defense.blocking and D.TurnToBlock and Defense.camAim and hrp then
        local flat = Vector3.new(Defense.camAim.X, hrp.Position.Y, Defense.camAim.Z)
        if (flat - hrp.Position).Magnitude > 0.05 then
            hrp.CFrame = CFrame.lookAt(hrp.Position, flat)
        end
    end
end)

-- ---- Nhan biet ky nang (phim) & danh thuong (chuot / Tool) ----
track(UserInputService.InputBegan:Connect(function(input)
    if UserInputService:GetFocusedTextBox() then return end
    if input.UserInputType == Enum.UserInputType.Keyboard then
        local skill = CONFIG.SkillKeys[input.KeyCode]
        if skill then
            AutoLock:trigger(skill.Duration or CONFIG.SkillDuration,
                             skill.Range or CONFIG.LockRange)
        end
    elseif input.UserInputType == Enum.UserInputType.MouseButton1 then
        AutoLock:trigger(CONFIG.BasicDuration, CONFIG.BasicRange)
    end
end))

local hookedTools = {}
local function hookTool(tool)
    if not tool:IsA("Tool") or hookedTools[tool] then return end
    hookedTools[tool] = true
    track(tool.Activated:Connect(function()
        AutoLock:trigger(CONFIG.BasicDuration, CONFIG.BasicRange)
    end))
    tool.AncestryChanged:Connect(function()
        if not tool:IsDescendantOf(game) then hookedTools[tool] = nil end
    end)
end
local function hookContainer(container)
    if not container then return end
    for _, t in ipairs(container:GetChildren()) do hookTool(t) end
    track(container.ChildAdded:Connect(hookTool))
end

hookContainer(LocalPlayer:WaitForChild("Backpack"))
if getCharacter() then hookContainer(getCharacter()) end
track(LocalPlayer.CharacterAdded:Connect(function(char)
    releaseTarget()
    AutoLock.weight = 0
    Defense.blocking = false
    Defense.reacting = false
    Defense.camWeight = 0
    Defense.dodgeUntil = 0
    hookContainer(char)
    hookContainer(LocalPlayer:FindFirstChildOfClass("Backpack"))
end))
track(LocalPlayer.CharacterRemoving:Connect(function()
    releaseTarget()
    AutoLock.weight = 0
    Defense.blocking = false
end))

local function spawnTestAttack(blockable, missOffset)
    local hrp = getHRP()
    if not hrp or not isAlive() then notify("Chua co nhan vat", true) return end
    local D = CONFIG.Defense
    local right, look = hrp.CFrame.RightVector, hrp.CFrame.LookVector
    local startPos = hrp.Position + right * 45
    local endPos   = hrp.Position - right * 45 + look * (missOffset or 0)

    local part = Instance.new("Part")
    part.Shape = Enum.PartType.Ball
    part.Size = Vector3.new(4, 4, 4)
    part.Material = Enum.Material.Neon
    part.Color = blockable and Color3.fromRGB(80, 160, 255) or Color3.fromRGB(255, 120, 40)
    part.Anchored = true
    part.CanCollide = false; part.CanTouch = false; part.CanQuery = false
    part.Position = startPos
    part:SetAttribute("Damage", blockable and 20 or 30)
    part:SetAttribute("Blockable", blockable)
    part.Parent = workspace
    CollectionService:AddTag(part, D.AttackTag)

    local dur, t, conn = 1.6, 0, nil
    conn = RunService.Heartbeat:Connect(function(dt)
        t += dt
        if not part.Parent or t >= dur then
            conn:Disconnect(); part:Destroy(); return
        end
        part.Position = startPos:Lerp(endPos, t / dur)
    end)
end

local function updateMenuPos()
    local ap = bubble.AbsolutePosition
    local vp = gui.AbsoluteSize
    local x = math.clamp(ap.X, 4, math.max(4, vp.X - MENU_W - 4))
    local below = ap.Y + BUBBLE + 6 + MENU_H <= vp.Y
    local y = below and (ap.Y + BUBBLE + 6) or (ap.Y - MENU_H - 6)
    menu.Position = UDim2.fromOffset(x, math.max(4, y))
end

local function toggleMenu()
    menu.Visible = not menu.Visible
    if menu.Visible then updateMenuPos() end
end

-- keo nut tron; bam nhe (khong keo) = mo/dong menu
do
    local dragging, moved, dragStart, startPos = false, false, nil, nil
    bubble.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1
            or i.UserInputType == Enum.UserInputType.Touch then
            dragging, moved = true, false
            dragStart, startPos = i.Position, bubble.Position
        end
    end)
    track(UserInputService.InputChanged:Connect(function(i)
        if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement
            or i.UserInputType == Enum.UserInputType.Touch) then
            local d = i.Position - dragStart
            if d.Magnitude > 6 then moved = true end
            if moved then
                bubble.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X,
                                            startPos.Y.Scale, startPos.Y.Offset + d.Y)
                if menu.Visible then updateMenuPos() end
            end
        end
    end))
    track(UserInputService.InputEnded:Connect(function(i)
        if dragging and (i.UserInputType == Enum.UserInputType.MouseButton1
            or i.UserInputType == Enum.UserInputType.Touch) then
            dragging = false
            if not moved then toggleMenu() end
        end
    end))
end
