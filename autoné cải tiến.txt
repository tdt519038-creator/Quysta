local player = game.Players.LocalPlayer
local RS = game.RunService
local ReplicatedStorage = game:GetService("ReplicatedStorage")

local lastPerfect
local forceDashEvent
local debounce =false

-- Tìm các đối tượng cần thiết một lần và lưu trữ lại
local function findRequiredObjects()
    local states = workspace:FindFirstChild("States")
    if not states then
        warn("States object not found in Workspace.")
        return nil
    end

    local events = ReplicatedStorage:FindFirstChild("Events")
    if not events then
        warn("Events folder not found in ReplicatedStorage.")
        return nil
    end

    forceDashEvent = events:FindFirstChild("ForceDash")
    if not forceDashEvent then
        warn("ForceDash event not found in Events folder.")
        return nil
    end

    return states
end

local states = findRequiredObjects()
if not states then
    warn("Failed to find required objects. Script will not run.")
    return
end

local function getState()
    return states:FindFirstChild(player.Name)
end

local function dodge(character)
    if debounce then
        return
    end

    debounce =true

    local perfect = character:GetAttribute("Perfect")
    if perfect == nil or perfect == lastPerfect then
        debounce =false
        return
    end

    local direction = character:GetAttribute("NeededDirection")
    if not (direction == "Left" or direction == "Right" or direction == "Backward" or direction == "Forward") then
        debounce =false
        return
    end

    local state = getState()
    if not state then
        debounce =false
        return
    end

    local occupied = state:FindFirstChild("Occupied")
    if not occupied then
        debounce =false
        return
    end

    local locked = occupied:FindFirstChild("LockedOn")
    local dashing = occupied:FindFirstChild("Dashing")
    if not locked or not locked.Value or (dashing and dashing.Value) then
        debounce =false
        return
    end

    local enemy = locked.Value
    if enemy:GetAttribute("IsFeinting") ==true or enemy:GetAttribute("LocalIsFeinting") ==true then
        debounce =false
        return
    end

    local root = character:FindFirstChild("HumanoidRootPart")
    local eroot = enemy:FindFirstChild("HumanoidRootPart")
    if root and eroot and (root.Position - eroot.Position).Magnitude > 14 then
        debounce =false
        return
    end

    lastPerfect = perfect
    forceDashEvent:Fire({Direction = direction})

    -- Thử thực hiện lại nếu không thành công
    task.delay(0.1, function()
        if player.Character == character and character:GetAttribute("Perfect") == perfect and (not dashing or not dashing.Value) then
            forceDashEvent:Fire({Direction = direction})
        end
        debounce =false
    end)
end

-- Hủy bỏ kết nối cũ nếu có
if getgenv().dodge then
    getgenv().dodge:Disconnect()
end

if getgenv().dodgeChar then
    getgenv().dodgeChar:Disconnect()
end

-- Kết nối lại sự kiện
getgenv().dodge = RS.Heartbeat:Connect(function()
    if player.Character then
        dodge(player.Character)
    end
end)

getgenv().dodgeChar = player.CharacterAdded:Connect(function(newCharacter)
    lastPerfect = nil
    debounce =false
    dodge(newCharacter)
end)
