local function formatArgs(args)
    local result = ""
    for i, v in ipairs(args) do
        if type(v) == "string" then
            result = result .. '"' .. tostring(v) .. '"'
        elseif type(v) == "userdata" then
            result = result .. v:GetFullName()
        else
            result = result .. tostring(v)
        end
        if i < #args then result = result .. ", " end
    end
    return result
end

local function logRemote(remote, method, args)
    local remoteName = remote.Name
    local path = remote:GetFullName()
    local argString = formatArgs(args)
    
    -- Tạo cấu trúc script chuẩn để bạn có thể dán vào tab chạy luôn
    local scriptToCopy = string.format("game.%s:%s(%s)", path, method, argString)
    
    local LogBtn = Instance.new("TextButton")
    LogBtn.Size = UDim2.new(1, -10, 0, 38)
    LogBtn.BackgroundColor3 = Color3.fromRGB(40, 40, 40)
    LogBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
    LogBtn.Text = " [" .. method .. "] " .. remoteName
    LogBtn.Font = Enum.Font.SourceSans
    LogBtn.TextSize = 14
    LogBtn.TextXAlignment = Enum.TextXAlignment.Left
    LogBtn.Parent = ScrollingFrame
    
    LogBtn.MouseButton1Click:Connect(function()
        -- Arceus X Neo hỗ trợ hàm setclipboard tiêu chuẩn
        if setclipboard then
            setclipboard(scriptToCopy)
            LogBtn.BackgroundColor3 = Color3.fromRGB(0, 150, 80)
            LogBtn.Text = " ĐÃ COPY THÀNH CÔNG!"
            task.delay(0.8, function()
                LogBtn.BackgroundColor3 = Color3.fromRGB(40, 40, 40)
                LogBtn.Text = " [" .. method .. "] " .. remoteName
            end)
        else
            LogBtn.Text = " Lỗi: Executor không hỗ trợ setclipboard!"
        end
    end)
end

-- Tiến hành bắt các lệnh gọi từ game lên máy chủ
local oldFireServer
oldFireServer = hookmetamethod(game, "__namecall", function(self, ...)
    local method = getnamecallmethod()
    local args = {...}
    
    if method == "FireServer" and self:IsA("RemoteEvent") then
        logRemote(self, "FireServer", args)
    elseif method == "InvokeServer" and self:IsA("RemoteFunction") then
        logRemote(self, "InvokeServer", args)
    end
    
    return oldFireServer(self, ...)
end)
