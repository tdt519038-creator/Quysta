local mt = getrawmetatable(game)
local oldNamecall = mt.__namecall
setreadonly(mt, false)

-- Hàm này dùng để dịch các tham số (arguments) thành dạng chữ để viết thành code
local function formatArgument(arg)
    local argType = typeof(arg)
    if argType == "string" then
        return '"' .. arg .. '"'
    elseif argType == "number" or argType == "boolean" then
        return tostring(arg)
    elseif argType == "Instance" then
        return "game." .. arg:GetFullName()
    elseif argType == "Vector3" then
        return "Vector3.new(" .. tostring(arg) .. ")"
    elseif argType == "Color3" then
        return "Color3.new(" .. tostring(arg) .. ")"
    else
        return "nil --[[ Loại dữ liệu chưa hỗ trợ: " .. argType .. " ]]"
    end
end

mt.__namecall = newcclosure(function(self, ...)
    local method = getnamecallmethod()
    local args = {...}

    -- Chỉ bắt FireServer (RemoteEvent) hoặc InvokeServer (RemoteFunction) do game gọi
    if not checkcaller() and (method == "FireServer" or method == "InvokeServer") then
        
        -- Tạo ra đoạn code giống hệt SimpleSpy
        local generatedCode = "\n--===== BẮT ĐƯỢC REMOTE: " .. self.Name .. " =====--\n"
        generatedCode = generatedCode .. "local args = {\n"
        
        for i, v in ipairs(args) do
            generatedCode = generatedCode .. "    [" .. i .. "] = " .. formatArgument(v) .. ",\n"
        end
        
        generatedCode = generatedCode .. "}\n\n"
        generatedCode = generatedCode .. "game." .. self:GetFullName() .. ":" .. method .. "(unpack(args))\n"
        generatedCode = generatedCode .. "--========================================--\n"
        
        -- In ra Console (Bấm F9 để xem hoặc xem ở bảng log của Executor)
        print(generatedCode)
    end

    return oldNamecall(self, ...)
end)

setreadonly(mt, true)
print("Minimal Spy đã chạy! Bấm F9 để xem code khi bạn click trong game.")
