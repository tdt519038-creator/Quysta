local state = {
    mannequin   = false,  
    vfxStrip    = false,  
    flatMat     = false,  
    noDecal     = false,  
    noShadow    = false,  
    lowTerrain  = false,  
    renderDist  = 500,    
    cullEnabled = false,  
    afkSaver    = false,

local ORIG = {}

-- ============================================================
-- TIEN ICH
-- ============================================================
local function eachCharacter(skipSelf, fn)
    for _, p in ipairs(Players:GetPlayers()) do
        if not (skipSelf and p == LocalPlayer) and p.Character then
            fn(p, p.Character)
        end
    end
end

-- 1) MANNEQUIN MODE
local strippedChars = {}

local function stripCharacter(character)
    if character == LocalPlayer.Character then return end
    for _, d in ipairs(character:GetDescendants()) do
        if d:IsA("Accessory") then
            if d:FindFirstChild("Handle") then
                d.Handle.LocalTransparencyModifier = 1
            end
            for _, p in ipairs(d:GetDescendants()) do
                if p:IsA("BasePart") then p.LocalTransparencyModifier = 1 end
            end
        elseif d:IsA("Shirt") or d:IsA("Pants") or d:IsA("ShirtGraphic") then
            d:Destroy()
        elseif d:IsA("BasePart") and d.Name ~= "HumanoidRootPart" then
            d.Color = Color3.fromRGB(150, 150, 150)
            d.Material = Enum.Material.SmoothPlastic
        elseif d:IsA("Decal") or d:IsA("SpecialMesh") then
            d.Transparency = 1
        end
    end
    strippedChars[character] = true
end

local function applyMannequin(on)
    state.mannequin = on
    eachCharacter(true, function(p, char)
        if on then stripCharacter(char) end
    end)
end

Players.PlayerAdded:Connect(function(p)
    p.CharacterAdded:Connect(function(char)
        if state.mannequin then task.wait(0.3) stripCharacter(char) end
    end)
end)

-- 2) VFX STRIPPER
local function filterVFX()
    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("ParticleEmitter") or obj:IsA("Trail") or obj:IsA("Beam") or obj:IsA("Sparkles") then
            local ownChar = obj:FindFirstAncestorWhichIsA("Model") == LocalPlayer.Character
            if not ownChar and not obj:GetAttribute("KeepVFX") then
                obj.Enabled = false
            end
        end
    end
end

local vfxConn
local function applyVFXStrip(on)
    state.vfxStrip = on
    if on then
        filterVFX()
        vfxConn = workspace.DescendantAdded:Connect(function(obj)
            if obj:IsA("ParticleEmitter") or obj:IsA("Trail") or obj:IsA("Beam") or obj:IsA("Sparkles") then
                task.wait(0.05)
                local ownChar = obj:FindFirstAncestorWhichIsA("Model") == LocalPlayer.Character
                if not ownChar and not obj:GetAttribute("KeepVFX") then
                    obj.Enabled = false
                end
            end
        end)
    elseif vfxConn then
        vfxConn:Disconnect()
        vfxConn = nil
    end
end

-- 3) FLAT MATERIAL
local flatBackup = {}
local function applyFlatMaterial(on)
    state.flatMat = on
    if on then
        for _, d in ipairs(workspace:GetDescendants()) do
            if d:IsA("BasePart") and d.Material ~= Enum.Material.SmoothPlastic then
                flatBackup[d] = d.Material
                d.Material = Enum.Material.SmoothPlastic
            end
        end
    else
        for part, mat in pairs(flatBackup) do
            if part.Parent then part.Material = mat end
        end
        flatBackup = {}
    end
end

-- 4) NO DECAL
local decalBackup = {}
local function applyNoDecal(on)
    state.noDecal = on
    if on then
        for _, d in ipairs(workspace:GetDescendants()) do
            if (d:IsA("Decal") or d:IsA("Texture")) and d.Transparency < 1 then
                decalBackup[d] = d.Transparency
                d.Transparency = 1
            end
        end
    else
        for d, t in pairs(decalBackup) do
            if d.Parent then d.Transparency = t end
        end
        decalBackup = {}
    end
end

-- 5) NO SHADOW & FOG
local function applyNoShadow(on)
    state.noShadow = on
    if on then
        ORIG.GlobalShadows = Lighting.GlobalShadows
        ORIG.FogEnd = Lighting.FogEnd
        ORIG.FogStart = Lighting.FogStart
        Lighting.GlobalShadows = false
        Lighting.FogEnd = 100000
        for _, e in ipairs(Lighting:GetChildren()) do
            if e:IsA("PostEffect") then e.Enabled = false end
        end
    else
        if ORIG.GlobalShadows ~= nil then Lighting.GlobalShadows = ORIG.GlobalShadows end
        if ORIG.FogEnd then Lighting.FogEnd = ORIG.FogEnd end
        if ORIG.FogStart then Lighting.FogStart = ORIG.FogStart end
    end
end

-- 6) LOW TERRAIN
local function applyLowTerrain(on)
    state.lowTerrain = on
    local terrain = workspace:FindFirstChildOfClass("Terrain")
    if not terrain then return end
    if on then
        ORIG.WaterWaveSize = terrain.WaterWaveSize
        ORIG.WaterReflect  = terrain.WaterReflectance
        ORIG.Decoration    = terrain.Decoration
        terrain.WaterWaveSize = 0
        terrain.WaterReflectance = 0
        terrain.Decoration = false
    else
        if ORIG.WaterWaveSize then terrain.WaterWaveSize = ORIG.WaterWaveSize end
        if ORIG.WaterReflect then terrain.WaterReflectance = ORIG.WaterReflect end
        if ORIG.Decoration ~= nil then terrain.Decoration = ORIG.Decoration end
    end
end

-- 7) CHUNK CULLING
local culled = {}
local function updateCulling()
    if not state.cullEnabled then return end
    local root = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not root then return end
    local pos = root.Position
    for _, m in ipairs(workspace:GetChildren()) do
        if m:IsA("Model") and m ~= LocalPlayer.Character and not Players:GetPlayerFromCharacter(m) then
            local ok, cf = pcall(function() return m:GetPivot() end)
            if ok then
                local dist = (cf.Position - pos).Magnitude
                local shouldShow = dist <= state.renderDist
                if culled[m] == nil or culled[m] ~= (not shouldShow) then
                    for _, part in ipairs(m:GetDescendants()) do
                        if part:IsA("BasePart") then
                            part.LocalTransparencyModifier = shouldShow and 0 or 1
                            part.CanQuery = shouldShow
                        end
                    end
                    culled[m] = not shouldShow
                end
            end
        end
    end
end

-- 8) AFK SAVER
local lastMoveTime = os.clock()
local afkApplied = false
local afkOrigQuality = nil

local function onCharMoved()
    lastMoveTime = os.clock()
    if afkApplied then
        afkApplied = false
        pcall(function()
            settings().Rendering.QualityLevel = afkOrigQuality or Enum.QualityLevel.Automatic
        end)
    end
end

-- 9) GARBAGE COLLECTION
local function gcNow()
    collectgarbage("collect")
end

task.spawn(function()
    while true do
        task.wait(60)
        gcNow()
    end
end)

-- LOOP
local acc = 0
RunService.Heartbeat:Connect(function(dt)
    acc += dt
    if acc < 0.5 then return end
    acc = 0
    if state.cullEnabled then updateCulling() end
    if state.afkSaver and not afkApplied and (os.clock() - lastMoveTime) > 20 then
        afkApplied = true
        pcall(function()
            afkOrigQuality = settings().Rendering.QualityLevel
            settings().Rendering.QualityLevel = Enum.QualityLevel.Level01
        end)
    end
end)

UIS.InputChanged:Connect(onCharMoved)
UIS.InputBegan:Connect(onCharMoved)

local dragging, dragStart, startPos = false, nil, nil
local dragDistance = 0

ball.InputBegan:Connect(function(io)
    if io.UserInputType == Enum.UserInputType.MouseButton1 or io.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = io.Position
        startPos = ball.Position
        dragDistance = 0
    end
end)

UIS.InputChanged:Connect(function(io)
    if dragging and (io.UserInputType == Enum.UserInputType.MouseMovement or io.UserInputType == Enum.UserInputType.Touch) then
        local delta = io.Position - dragStart
        dragDistance = delta.Magnitude
        ball.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X,
                                   startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        panel.Position = UDim2.new(0, ball.Position.X.Offset + 64, 0, ball.Position.Y.Offset - 190)
    end
end)

UIS.InputEnded:Connect(function(io)
    if io.UserInputType == Enum.UserInputType.MouseButton1 or io.UserInputType == Enum.UserInputType.Touch then
        if dragging then
            if dragDistance < 10 then -- Nếu lệch ít hơn 10px thì coi như là bấm mở menu
                panel.Visible = not panel.Visible
            end
            dragging = false
        end
    end
end)
