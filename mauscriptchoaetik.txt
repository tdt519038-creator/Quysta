local function getCharacter() return LocalPlayer.Character end
local function getHumanoid()
    local c = getCharacter()
    return c and c:FindFirstChildOfClass("Humanoid") or nil
end
local function getHRP()
    local c = getCharacter()
    return c and c:FindFirstChild("HumanoidRootPart") or nil
end
local function isAlive()
    local h = getHumanoid()
    return h ~= nil and h.Health > 0
end

local NotifyContainer
local function notify(text, isError)
    if not NotifyContainer or not NotifyContainer.Parent then return end
    local frame = Instance.new("Frame")
    frame.BackgroundColor3 = isError and Color3.fromRGB(180,60,60) or Color3.fromRGB(40,45,55)
    frame.BackgroundTransparency = 0.1
    frame.Size = UDim2.new(0, 240, 0, 30)
    frame.AutomaticSize = Enum.AutomaticSize.X
    frame.Position = UDim2.new(0.5, 0, 0, 0)
    frame.AnchorPoint = Vector2.new(0.5, 0)
    frame.BorderSizePixel = 0
    local c = Instance.new("UICorner"); c.CornerRadius = UDim.new(0,6); c.Parent = frame
    local l = Instance.new("TextLabel")
    l.BackgroundTransparency = 1
    l.Size = UDim2.new(1,-16,1,0); l.Position = UDim2.new(0,8,0,0)
    l.Font = Enum.Font.GothamMedium; l.TextSize = 12
    l.TextColor3 = Color3.new(1,1,1); l.TextXAlignment = Enum.TextXAlignment.Left
    l.Text = text; l.Parent = frame
    frame.Parent = NotifyContainer
    task.spawn(function()
        task.wait(CONFIG.NotifyDuration)
        if frame.Parent then
            TweenService:Create(frame, TweenInfo.new(0.3), {BackgroundTransparency=1}):Play()
            TweenService:Create(l, TweenInfo.new(0.3), {TextTransparency=1}):Play()
            task.wait(0.35); frame:Destroy()
        end
    end)
    for _, o in ipairs(NotifyContainer:GetChildren()) do
        if o:IsA("Frame") and o ~= frame then
            local y = o.Position.Y.Offset
            TweenService:Create(o, TweenInfo.new(0.2),
                {Position = UDim2.new(0.5,0,0,y-38)}):Play()
        end
    end
end

-- TIM QUAI (tag hoac folder)
-- ============================================================
local function getEnemyRoot(model)
    if not model or not model:IsA("Model") then return nil end
    return model:FindFirstChild("HumanoidRootPart")
        or model.PrimaryPart
        or model:FindFirstChildWhichIsA("BasePart", true)
end

local function isValidEnemy(model)
    if not model or not model.Parent then return false end
    if not model:IsA("Model") then return false end
    if model == getCharacter() then return false end
    if Players:GetPlayerFromCharacter(model) then return false end
    local hum = model:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 then return false end
    return getEnemyRoot(model) ~= nil
end

local function collectEnemies()
    local list, seen = {}, {}
    local function add(m)
        if not seen[m] and isValidEnemy(m) then
            seen[m] = true
            table.insert(list, m)
        end
    end
    for _, m in ipairs(CollectionService:GetTagged(CONFIG.EnemyTag)) do add(m) end
    local folder = workspace:FindFirstChild(CONFIG.FallbackEnemyFolder)
    if folder then
        for _, m in ipairs(folder:GetChildren()) do add(m) end
    end
    return list
end

local function findNearestEnemy(range)
    local hrp = getHRP()
    if not hrp then return nil end
    local best, bestDist = nil, range
    for _, m in ipairs(collectEnemies()) do
        local root = getEnemyRoot(m)
        local d = (root.Position - hrp.Position).Magnitude
        if d <= bestDist then
            best, bestDist = m, d
        end
    end
    return best
end

-- ============================================================
-- AUTO LOCK-ON
-- ============================================================
local AutoLock = {
    enabled = true,
    target = nil,
    expire = 0,
    weight = 0,            -- 0 = camera binh thuong, 1 = ghim hoan toan
    lastAimPos = nil,
    savedAutoRotate = nil,
    highlight = nil,
}

local function setHighlight(model)
    if AutoLock.highlight then
        AutoLock.highlight:Destroy()
        AutoLock.highlight = nil
    end
    if model then
        local h = Instance.new("Highlight")
        h.FillTransparency = 1
        h.OutlineColor = Color3.fromRGB(255, 60, 60)
        h.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
        h.Parent = model
        AutoLock.highlight = h
    end
end

local function restoreAutoRotate()
    if AutoLock.savedAutoRotate ~= nil then
        local hum = getHumanoid()
        if hum then hum.AutoRotate = AutoLock.savedAutoRotate end
        AutoLock.savedAutoRotate = nil
    end
end

local function releaseTarget()
    AutoLock.target = nil
    AutoLock.expire = 0
    setHighlight(nil)
    restoreAutoRotate()
end

-- Goi khi danh thuong / dung ky nang
function AutoLock:trigger(duration)
    if not self.enabled or not isAlive() then return end
    local target = findNearestEnemy(CONFIG.LockRange)
    if not target then return end            -- khong co quai trong tam -> khong ghim
    if target ~= self.target then
        self.target = target
        setHighlight(target)
    end
    self.expire = os.clock() + duration
end

RunService:BindToRenderStep("AutoLockCamera", Enum.RenderPriority.Camera.Value + 1, function(dt)
    local cam = workspace.CurrentCamera
    local hrp = getHRP()
    local active = false

    local target = AutoLock.target
    if target then
        local root = isValidEnemy(target) and getEnemyRoot(target) or nil
        if root and hrp and isAlive() and os.clock() < AutoLock.expire
            and (root.Position - hrp.Position).Magnitude <= CONFIG.LockRange * CONFIG.LoseRangeMul then
            active = true
            AutoLock.lastAimPos = root.Position

            -- Xoay nhan vat ve muc tieu (chi truc ngang)
            if CONFIG.TurnCharacter then
                local hum = getHumanoid()
                if hum then
                    if AutoLock.savedAutoRotate == nil then
                        AutoLock.savedAutoRotate = hum.AutoRotate
                    end
                    hum.AutoRotate = false
                end
                local flat = Vector3.new(root.Position.X, hrp.Position.Y, root.Position.Z)
                if (flat - hrp.Position).Magnitude > 0.05 then
                    hrp.CFrame = CFrame.lookAt(hrp.Position, flat)
                end
            end
        else
            -- Het thoi gian ky nang / muc tieu chet / ra khoi tam
            releaseTarget()
        end
    end

    -- Blend camera: vao muc tieu khi active, tra ve binh thuong khi het
    if active then
        AutoLock.weight = math.min(1, AutoLock.weight + dt / CONFIG.CamBlendIn)
    else
        AutoLock.weight = math.max(0, AutoLock.weight - dt / CONFIG.CamBlendOut)
    end

    if AutoLock.weight > 0 and AutoLock.lastAimPos and cam then
        local cf = cam.CFrame
        if (AutoLock.lastAimPos - cf.Position).Magnitude > 0.1 then
            local look = CFrame.lookAt(cf.Position, AutoLock.lastAimPos)
            -- cf la CFrame do camera mac dinh vua tinh -> lerp ve 0 = camera binh thuong
            cam.CFrame = cf:Lerp(look, AutoLock.weight)
        end
    end
end)

-- ---- Nhan biet ky nang (phim) & danh thuong (chuot / Tool) ----
local skillKeySet = {}
for _, k in ipairs(CONFIG.SkillKeys) do skillKeySet[k] = true end

UserInputService.InputBegan:Connect(function(input)
    if UserInputService:GetFocusedTextBox() then return end
    if input.UserInputType == Enum.UserInputType.Keyboard then
        if skillKeySet[input.KeyCode] then
            AutoLock:trigger(CONFIG.SkillDuration)
        end
    elseif input.UserInputType == Enum.UserInputType.MouseButton1 then
        AutoLock:trigger(CONFIG.BasicDuration)
    end
end)

-- Tool.Activated (ho tro danh thuong tren mobile / game dung Tool)
local hookedTools = {}
local function hookTool(tool)
    if not tool:IsA("Tool") or hookedTools[tool] then return end
    hookedTools[tool] = true
    tool.Activated:Connect(function()
        AutoLock:trigger(CONFIG.BasicDuration)
    end)
    tool.AncestryChanged:Connect(function()
        if not tool:IsDescendantOf(game) then hookedTools[tool] = nil end
    end)
end
local function hookContainer(container)
    if not container then return end
    for _, t in ipairs(container:GetChildren()) do hookTool(t) end
    container.ChildAdded:Connect(hookTool)
end

hookContainer(LocalPlayer:WaitForChild("Backpack"))
if getCharacter() then hookContainer(getCharacter()) end
LocalPlayer.CharacterAdded:Connect(function(char)
    releaseTarget()
    AutoLock.weight = 0
    hookContainer(char)
    hookContainer(LocalPlayer:FindFirstChildOfClass("Backpack"))
end)
LocalPlayer.CharacterRemoving:Connect(function()
    releaseTarget()
    AutoLock.weight = 0
end)

    title.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1
            or i.UserInputType == Enum.UserInputType.Touch then
            dragging, dragStart, startPos = true, i.Position, main.Position
        end
    end)
    UserInputService.InputChanged:Connect(function(i)
        if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement
            or i.UserInputType == Enum.UserInputType.Touch) then
            local d = i.Position - dragStart
            main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X,
                                      startPos.Y.Scale, startPos.Y.Offset + d.Y)
        end
    end)
    UserInputService.InputEnded:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1
            or i.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)

    lbl.TextXAlignment = Enum.TextXAlignment.Left

    local box = Instance.new("TextBox")
    box.Size = UDim2.new(0.4, -12, 0, 26)
    box.Position = UDim2.new(0.6, 4, 0, y + 1)
    box.BackgroundColor3 = Color3.fromRGB(45, 49, 58)
    box.BorderSizePixel = 0
    box.Font = Enum.Font.GothamMedium
    box.TextSize = 12
    box.TextColor3 = Color3.new(1, 1, 1)
    box.ClearTextOnFocus = false
    box.Text = tostring(CONFIG[key])
    box.Parent = main
    Instance.new("UICorner", box).CornerRadius = UDim.new(0, 5)

    box.FocusLost:Connect(function()
        local n = tonumber(box.Text)
        if n and n > 0 then
            CONFIG[key] = n
        else
            notify("Gia tri khong hop le", true)
        end
        box.Text = tostring(CONFIG[key])
    end)
end
