local Client = game.Players.LocalPlayer
local States = workspace:WaitForChild("States")
local Effect = require(game:GetService("ReplicatedStorage").Modules.EffectHelper)
local VIM = game:GetService("VirtualInputManager")

local function hold(keyCode, time)
    set_thread_identity(7)
    VIM:SendKeyEvent(true, keyCode, false, game)
    task.wait(time)
    VIM:SendKeyEvent(false, keyCode, false, game)
    set_thread_identity(2)
end

local function get(n, s)
    local pState = States:FindFirstChild(n.Name)
    if pState then
        local target = pState:FindFirstChild(s, true)
        return target and target.Value
    end
    return nil
end

local function get_root(chr)
    return chr and chr:FindFirstChild("HumanoidRootPart")
end

local function notif(str, str2)
    game:GetService("StarterGui"):SetCore("SendNotification", {
        Title = str,
        Text = str2,
        Duration = 2
    })
end

local function dash()
    hold(Enum.KeyCode.Space, 0.1) 
end

local waiting = {}

local function waitValid(name, func)
    if not waiting[name] then 
        waiting[name] = {} 
    end
    table.insert(waiting[name], func)
    warn("Added Queue", name)
end

local function valid(name, ...)
    if waiting[name] then
        for _, func in ipairs(waiting[name]) do
            task.spawn(func, ...)
        end
        waiting[name] = nil
        warn("Cleared Queue", name)
    end
end

        -- KHẲNG ĐỊNH: Bật lên sẽ chạy chuẩn cấu trúc kiểm tra để né chiêu đàng hoàng
        if type(d) == "table" and typeof(d[2]) == "Instance" then
            local enemyChar = d[2]
            local localRoot = get_root(Client.Character)
            local enemyRoot = get_root(enemyChar)
            
            if localRoot and enemyRoot then
                local Distance = Client:DistanceFromCharacter(enemyRoot.Position)
                
                local isBlocking = get(Client, "Blocking")
                local isEquipped = get(Client, "Equipped")
                local isPunching = get(Client, "Punching")
                
                if not isBlocking and isEquipped and not isPunching then
                    if Distance <= 10.8 and (d[1] == "AttackTrail" or d[1] == "StartupHighlight" or d[1] == "UltimateHighlight") and d[2] ~= Client.Character then
                        
                        if type(d[5]) == 'number' then
                            waitValid(d[2].Name, function()
                                local randomDelay = Random.new():NextNumber(0.4, 0.6)
                                local formattedResult = tonumber(string.format("%.2f", randomDelay))
                                
                                task.wait(formattedResult)
                                task.spawn(dash)
                                notif("Dodged!", "Truly a master at work")
                            end)

                    end
                end
            end
            
            if d[1] == "StartupHighlight" then

            end
        end
        
        return shared.BaseEffectFunction[i](d, ...)
    end
end
